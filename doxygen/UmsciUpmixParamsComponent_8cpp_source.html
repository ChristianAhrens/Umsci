21UmsciUpmixParamsComponent::UmsciUpmixParamsComponent()
22{
23 setOpaque(false);
24 setInterceptsMouseClicks(true, true);
25
26 m_spreadSlider = std::make_unique<juce::Slider>(juce::Slider::LinearHorizontal,
27 juce::Slider::NoTextBox);
28 m_spreadSlider->setRange(0.0, 1.0, 0.0);
29 m_spreadSlider->setValue(static_cast<double>(m_spread), juce::dontSendNotification);
30 m_spreadSlider->setColour(juce::Slider::thumbColourId, m_highlightColour);
31 m_spreadSlider->setColour(juce::Slider::trackColourId, m_highlightColour.withAlpha(0.5f));
32 m_spreadSlider->onValueChange = [this] {
33 if (m_suppressCallbacks) return;
34 m_spread = static_cast<float>(m_spreadSlider->getValue());
35 checkMismatch();
36 if (onSpreadChanged) onSpreadChanged(m_spread);
37 };
38 addAndMakeVisible(m_spreadSlider.get());
39
40 auto initButton = [this](std::unique_ptr<juce::TextButton>& btn, const juce::String& label) {
41 btn = std::make_unique<juce::TextButton>(label);
42 btn->setClickingTogglesState(false);
43 btn->setColour(juce::TextButton::buttonColourId, juce::Colours::transparentBlack);
44 btn->setColour(juce::TextButton::buttonOnColourId, m_highlightColour);
45 addAndMakeVisible(btn.get());
46 };
47 initButton(m_delayModeOff, "Off");
48 initButton(m_delayModeTight, "Tight");
49 initButton(m_delayModeFull, "Full");
50
51 m_delayModeOff->onClick = [this] {
52 if (m_suppressCallbacks) return;
53 m_delayMode = 0;
54 updateDelayModeButtonStates();
55 checkMismatch();
56 if (onDelayModeChanged) onDelayModeChanged(0);
57 };
58 m_delayModeTight->onClick = [this] {
59 if (m_suppressCallbacks) return;
60 m_delayMode = 1;
61 updateDelayModeButtonStates();
62 checkMismatch();
63 if (onDelayModeChanged) onDelayModeChanged(1);
64 };
65 m_delayModeFull->onClick = [this] {
66 if (m_suppressCallbacks) return;
67 m_delayMode = 2;
68 updateDelayModeButtonStates();
69 checkMismatch();
70 if (onDelayModeChanged) onDelayModeChanged(2);
71 };
72 updateDelayModeButtonStates();
73
74 if (m_syncProblemDrawable = juce::Drawable::createFromSVGString(BinaryData::sync_problem_24dp_svg))
75 m_syncProblemDrawable->replaceColour(juce::Colours::black, m_highlightColour);
76}

254void UmsciUpmixParamsComponent::resized()
255{
256 const int pad = getPadding();
257 const int titleH = getTitleRowHeight();
258 const int rowH = getContentRowHeight();
259 const int gap = rowH / 4;
260 const int contentW = getWidth() - getGrabStripWidth();
261 const int innerW = contentW - 2 * pad;
262 const int innerX = pad;
263 const int labelW = innerW / 4;
264
265 // Row 1: "Spread" label (painted) + slider — gap below title
266 m_spreadSlider->setBounds(innerX + labelW, pad + titleH + gap, innerW - labelW, rowH);
267
268 // Row 2: "Delay:" label (painted) + three equal-width buttons — gap below row 1
269 const int btnAreaW = innerW - labelW;
270 const int btnW = btnAreaW / 3;
271 const int btnX = innerX + labelW;
272 const int btnY = pad + titleH + gap + rowH + gap;
273 m_delayModeOff->setBounds(btnX, btnY, btnW, rowH);
274 m_delayModeTight->setBounds(btnX + btnW, btnY, btnW, rowH);
275 m_delayModeFull->setBounds(btnX + 2 * btnW, btnY, btnAreaW - 2 * btnW, rowH);
276}

307void UmsciUpmixParamsComponent::paintGrabStrip(juce::Graphics& g, juce::Rectangle<int> stripBounds)
308{
309 constexpr float pillW = 4.0f;
310 constexpr float pillH = 2.0f;
311 constexpr float gap = 5.0f;
312
313 const float cx = static_cast<float>(stripBounds.getCentreX());
314 const float cy = static_cast<float>(stripBounds.getCentreY());
315
316 g.setColour(m_highlightColour.withAlpha(0.75f));
317 for (int i = -1; i <= 1; ++i)
318 {
319 const auto pill = juce::Rectangle<float>(
320 cx - pillW * 0.5f,
321 cy + static_cast<float>(i) * gap - pillH * 0.5f,
322 pillW, pillH);
323 g.fillRoundedRectangle(pill, pillH * 0.5f);
324 }
325}

327void UmsciUpmixParamsComponent::mouseUp(const juce::MouseEvent& e)
328{
329 const int grabX = getWidth() - getGrabStripWidth();
330
331 if (e.x >= grabX)
332 {
333 // Grab strip toggles panel state (always reachable).
334 if (onStateChangeRequested)
335 {
336 const auto newState = (m_state == PanelState::Tucked) ? PanelState::Visible : PanelState::Tucked;
337 onStateChangeRequested(newState);
338 }
339 }
340 else if (isTimerRunning())
341 {
342 // Content area click while out of sync: push current aggregate values to device.
343 // Children are blocked (setInterceptsMouseClicks false) so every click lands here.
344 if (onSpreadChanged) onSpreadChanged(m_spread);
345 if (onDelayModeChanged) onDelayModeChanged(m_delayMode);
346 }
347}

358void UmsciUpmixParamsComponent::checkMismatch()
359{
360 constexpr float kSpreadTol = 0.01f;
361 bool mismatch = false;
362
363 for (auto id : m_monitoredSourceIds)
364 {
365 auto spreadIt = m_deviceSpread.find(id);
366 if (spreadIt == m_deviceSpread.end()
367 || std::abs(spreadIt->second - m_spread) > kSpreadTol)
368 {
369 mismatch = true;
370 break;
371 }
372 auto delayIt = m_deviceDelayMode.find(id);
373 if (delayIt == m_deviceDelayMode.end()
374 || static_cast<int>(delayIt->second) != m_delayMode)
375 {
376 mismatch = true;
377 break;
378 }
379 }
380
381 if (mismatch && !isTimerRunning())
382 {
383 startTimer(500);
384 setChildrenMouseInterception(false);
385 }
386 else if (!mismatch && isTimerRunning())
387 {
388 stopTimer();
389 m_flashState = false;
390 setAlpha(1.0f);
391 setChildrenMouseInterception(true);
392 repaint();
393 }
394}

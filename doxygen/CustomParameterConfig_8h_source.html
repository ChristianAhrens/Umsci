40 std::vector<std::string> stepNames;
41 juce::String oscAddress = "/param/0";
42
44 JUCEAppBasics::ParameterControlInfo toParameterControlInfo(int index) const
45 {
46 JUCEAppBasics::ParameterControlInfo info;
47 info.index = index;
48 info.name = name;
49 info.type = controlType;
50 info.minValue = minValue;
51 info.maxValue = maxValue;
52 info.stepSize = stepSize;
53 info.stepCount = stepCount;
54 info.stepNames = stepNames;
55 info.currentValue = minValue; // initialised to minimum

57 }

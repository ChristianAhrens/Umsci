102
107 struct RemObjAddr
108 {
109 std::int16_t pri{0};
110 std::int16_t sec{0};
111
112 static constexpr std::int16_t sc_INV = 0;
113
114 RemObjAddr() = default;
115 RemObjAddr(const RemObjAddr&) = default;
116 RemObjAddr(std::int16_t a, std::int16_t b) : pri(a), sec(b) {}
117
118 juce::String toNiceString() const
119 {
120 if (sc_INV != pri && sc_INV != sec)
121 return juce::String(sec) + "/" + juce::String(pri);
122 else if (sc_INV != pri)
123 return juce::String(pri);
124 else if (sc_INV != sec)
125 return juce::String(sec);
126 else
127 return {};
128 }
129 juce::String toString() const
130 {
131 return juce::String(pri) + "," + juce::String(sec);
132 }
133 static juce::String toString(const std::vector<RemObjAddr>& addrs)
134 {
135 juce::String s;
136 for (auto const& a : addrs)
137 s << a.toString() << ";";
138 return s;
139 }
140 bool fromString(const juce::String& csv)
141 {
142 juce::StringArray sa;
143 if (sa.addTokens(csv, ",", "") != 2 || sa.size() != 2)
144 return false;
145 pri = std::int16_t(sa[0].getIntValue());
146 sec = std::int16_t(sa[1].getIntValue());
147 return true;
148 }
149 static RemObjAddr createFromString(const juce::String& csv)
150 {
151 juce::StringArray sa;
152 sa.addTokens(csv.trimCharactersAtEnd(","), ",", "");
153 if (sa.size() != 2)
154 return {};
155 return { std::int16_t(sa[0].getIntValue()), std::int16_t(sa[1].getIntValue()) };
156 }
157 static std::vector<RemObjAddr> createFromListString(const juce::String& list)
158 {
159 std::vector<RemObjAddr> result;
160 juce::StringArray sa;
161 sa.addTokens(list.trimCharactersAtEnd(";"), ";", "");
162 for (auto const& csv : sa)
163 result.push_back(createFromString(csv));
164 return result;
165 }
166
167 bool operator==(const RemObjAddr& r) const { return pri == r.pri && sec == r.sec; }
168 bool operator!=(const RemObjAddr& r) const { return !(*this == r); }
169 bool operator<(const RemObjAddr& r) const
170 {
171 return pri < r.pri || (pri == r.pri && sec < r.sec);
172 }
173 bool operator>(const RemObjAddr& r) const { return !(*this < r) && (*this != r); }
174 RemObjAddr& operator=(const RemObjAddr&) = default;
175
176 JUCE_LEAK_DETECTOR(RemObjAddr)
177 };

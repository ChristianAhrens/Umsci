92void CustomParameterControlConfigDialog::resized()
93{
94 auto bounds = getLocalBounds().reduced(kMargin);
95 bounds.removeFromTop(18); // section label "Connection"
96
97 // Row 1: host
98 auto connRow1 = bounds.removeFromTop(kRowH);
99 auto hostLabelBounds = connRow1.removeFromLeft(90);
100 m_hostLabel->setBounds(hostLabelBounds);
101 m_hostEditor->setBounds(connRow1.removeFromLeft(180).reduced(0, 2));
102
103 // Row 2: ports
104 auto connRow2 = bounds.removeFromTop(kRowH);
105 connRow2.removeFromLeft(90); // align under host editor
106 auto sendPortLabelBounds = connRow2.removeFromLeft(72);
107 m_sendPortLabel->setBounds(sendPortLabelBounds);
108 m_sendPortEditor->setBounds(connRow2.removeFromLeft(70).reduced(0, 2));
109 connRow2.removeFromLeft(8);
110 auto recvPortLabelBounds = connRow2.removeFromLeft(84);
111 m_recvPortLabel->setBounds(recvPortLabelBounds);
112 m_recvPortEditor->setBounds(connRow2.removeFromLeft(70).reduced(0, 2));
113
114 bounds.removeFromTop(kMargin); // gap before "Parameters" label
115 bounds.removeFromTop(18); // section label "Parameters"
116
117 // Column header row heights (shared with content rows)
118 const int kViewportH = juce::jmax(kRowH, getHeight() - kHeaderH - kAddBtnH - 3 * kMargin - 18);
119 m_viewport->setBounds(bounds.removeFromTop(kViewportH));
120
121 bounds.removeFromTop(kMargin);
122 m_addButton->setBounds(bounds.removeFromTop(kAddBtnH));
123
124 // Layout list content rows
125 if (m_listContent)
126 {
127 int contentH = int(m_paramRows.size()) * (kRowH + 2);
128 m_listContent->setSize(m_viewport->getWidth() - m_viewport->getScrollBarThickness(),
129 juce::jmax(kRowH, contentH));
130
131 // Column widths — variable section (min+max+step or stepNames) occupies the same total
132 // space so the OSC address column is always the same width regardless of type.
133 const int delW = 26;
134 const int typeW = 112;
135 const int nameW = 100;
136 const int minW = 52;
137 const int maxW = 52;
138 const int stepW = 52;
139 const int gap = 3;
140 const int varSectionW = minW + maxW + stepW + 2 * gap; // Continuous: min+gap+max+gap+step
141
142 int y = 0;
143 for (int i = 0; i < int(m_paramRows.size()); ++i)
144 {
145 auto& row = m_paramRows[i];
146 auto rowBounds = juce::Rectangle<int>(0, y, m_listContent->getWidth(), kRowH);
147 y += kRowH + 2;
148
149 auto r = rowBounds.reduced(0, 2);
150 row.deleteButton->setBounds(r.removeFromLeft(delW)); r.removeFromLeft(gap);
151 row.typeCombo->setBounds(r.removeFromLeft(typeW)); r.removeFromLeft(gap);
152 row.nameEditor->setBounds(r.removeFromLeft(nameW)); r.removeFromLeft(gap);
153
154 if (row.minEditor->isVisible())
155 {
156 row.minEditor->setBounds(r.removeFromLeft(minW)); r.removeFromLeft(gap);
157 row.maxEditor->setBounds(r.removeFromLeft(maxW)); r.removeFromLeft(gap);
158 row.stepEditor->setBounds(r.removeFromLeft(stepW)); r.removeFromLeft(gap);
159 }
160 if (row.stepNamesEditor->isVisible())
161 {
162 row.stepNamesEditor->setBounds(r.removeFromLeft(varSectionW)); r.removeFromLeft(gap);
163 }
164
165 row.oscAddrEditor->setBounds(r.removeFromLeft(juce::jmax(50, r.getWidth())));
166 }
167 }
168}

183CustomParameterConfig CustomParameterControlConfigDialog::getConfig() const
184{
185 CustomParameterConfig result;
186 result.oscRemoteHost = m_hostEditor->getText().trim();
187 result.oscSendPort = m_sendPortEditor->getText().getIntValue();
188 result.oscReceivePort = m_recvPortEditor->getText().getIntValue();
189
190 for (int i = 0; i < int(m_paramRows.size()); ++i)
191 {
192 auto& row = m_paramRows[i];
193 CustomParameterEntry entry;
194
195 int typeIdx = row.typeCombo->getSelectedId() - 1;
196 if (typeIdx == 0) entry.controlType = JUCEAppBasics::ParameterControlType::Continuous;
197 else if (typeIdx == 1) entry.controlType = JUCEAppBasics::ParameterControlType::Discrete;
198 else entry.controlType = JUCEAppBasics::ParameterControlType::Toggle;
199
200 entry.name = row.nameEditor->getText().trim();
201 entry.oscAddress = row.oscAddrEditor->getText().trim();
202
203 switch (entry.controlType)
204 {
205 case JUCEAppBasics::ParameterControlType::Continuous:
206 entry.minValue = row.minEditor->getText().getFloatValue();
207 entry.maxValue = row.maxEditor->getText().getFloatValue();
208 entry.stepSize = row.stepEditor->getText().getFloatValue();
209 break;
210 case JUCEAppBasics::ParameterControlType::Discrete:
211 {
212 juce::StringArray names;
213 names.addTokens(row.stepNamesEditor->getText(), ",", "");
214 for (auto& n : names)
215 entry.stepNames.push_back(n.trim().toStdString());
216 entry.stepCount = int(entry.stepNames.size());
217 entry.minValue = 0.0f;
218 entry.maxValue = entry.stepCount > 1 ? float(entry.stepCount - 1) : 0.0f;
219 entry.stepSize = 1.0f;
220 break;
221 }
222 case JUCEAppBasics::ParameterControlType::Toggle:
223 entry.minValue = 0.0f;
224 entry.maxValue = 1.0f;
225 entry.stepSize = 1.0f;
226 break;
227 }
228
229 result.parameters.push_back(entry);
230 }
231
232 return result;
233}

235void CustomParameterControlConfigDialog::rebuildParameterRows()
236{
237 // Remove all existing row widgets from the list content
238 for (auto& row : m_paramRows)
239 {
240 if (m_listContent) m_listContent->removeChildComponent(row.typeCombo.get());
241 if (m_listContent) m_listContent->removeChildComponent(row.nameEditor.get());
242 if (m_listContent) m_listContent->removeChildComponent(row.minEditor.get());
243 if (m_listContent) m_listContent->removeChildComponent(row.maxEditor.get());
244 if (m_listContent) m_listContent->removeChildComponent(row.stepEditor.get());
245 if (m_listContent) m_listContent->removeChildComponent(row.stepNamesEditor.get());
246 if (m_listContent) m_listContent->removeChildComponent(row.oscAddrEditor.get());
247 if (m_listContent) m_listContent->removeChildComponent(row.deleteButton.get());
248 }
249 m_paramRows.clear();
250
251 for (int i = 0; i < int(m_config.parameters.size()); ++i)
252 addParameterRow(i);
253}

255void CustomParameterControlConfigDialog::addParameterRow(int rowIndex)
256{
257 if (rowIndex < 0 || rowIndex >= int(m_config.parameters.size()))
258 return;
259
260 const auto& entry = m_config.parameters[rowIndex];
261 ParamRow row;
262
263 row.typeCombo = std::make_unique<juce::ComboBox>();
264 row.typeCombo->addItem("Continuous", 1);
265 row.typeCombo->addItem("Discrete", 2);
266 row.typeCombo->addItem("Toggle", 3);
267 int typeId = 1;
268 if (entry.controlType == JUCEAppBasics::ParameterControlType::Discrete) typeId = 2;
269 else if (entry.controlType == JUCEAppBasics::ParameterControlType::Toggle) typeId = 3;
270 row.typeCombo->setSelectedId(typeId, juce::dontSendNotification);
271 row.typeCombo->onChange = [this, rowIndex]() { updateRowVisibility(rowIndex); };
272
273 row.nameEditor = std::make_unique<juce::TextEditor>();
274 row.nameEditor->setText(entry.name, juce::dontSendNotification);
275 row.nameEditor->setTextToShowWhenEmpty("Name", juce::Colours::grey);
276
277 row.minEditor = std::make_unique<juce::TextEditor>();
278 row.minEditor->setText(juce::String(entry.minValue), juce::dontSendNotification);
279 row.minEditor->setTextToShowWhenEmpty("min", juce::Colours::grey);
280
281 row.maxEditor = std::make_unique<juce::TextEditor>();
282 row.maxEditor->setText(juce::String(entry.maxValue), juce::dontSendNotification);
283 row.maxEditor->setTextToShowWhenEmpty("max", juce::Colours::grey);
284
285 row.stepEditor = std::make_unique<juce::TextEditor>();
286 row.stepEditor->setText(entry.stepSize > 0.0f ? juce::String(entry.stepSize) : "",
287 juce::dontSendNotification);
288 row.stepEditor->setTextToShowWhenEmpty("step", juce::Colours::grey);
289
290 row.stepNamesEditor = std::make_unique<juce::TextEditor>();
291 juce::StringArray namesArr;
292 for (auto& n : entry.stepNames) namesArr.add(n);
293 row.stepNamesEditor->setText(namesArr.joinIntoString(","), juce::dontSendNotification);
294 row.stepNamesEditor->setTextToShowWhenEmpty("step names (csv)", juce::Colours::grey);
295
296 row.oscAddrEditor = std::make_unique<juce::TextEditor>();
297 row.oscAddrEditor->setText(entry.oscAddress, juce::dontSendNotification);
298 row.oscAddrEditor->setTextToShowWhenEmpty("/osc/address", juce::Colours::grey);
299
300 const int capturedRow = rowIndex;
301 row.deleteButton = std::make_unique<juce::TextButton>("X");
302 row.deleteButton->onClick = [this, capturedRow]() {
303 removeParameterRow(capturedRow);
304 };
305
306 m_listContent->addAndMakeVisible(row.typeCombo.get());
307 m_listContent->addAndMakeVisible(row.nameEditor.get());
308 m_listContent->addAndMakeVisible(row.minEditor.get());
309 m_listContent->addAndMakeVisible(row.maxEditor.get());
310 m_listContent->addAndMakeVisible(row.stepEditor.get());
311 m_listContent->addAndMakeVisible(row.stepNamesEditor.get());
312 m_listContent->addAndMakeVisible(row.oscAddrEditor.get());
313 m_listContent->addAndMakeVisible(row.deleteButton.get());
314
315 m_paramRows.push_back(std::move(row));
316 updateRowVisibility(rowIndex);
317}

319void CustomParameterControlConfigDialog::updateRowVisibility(int rowIndex)
320{
321 if (rowIndex < 0 || rowIndex >= int(m_paramRows.size()))
322 return;
323
324 auto& row = m_paramRows[rowIndex];
325 const int typeId = row.typeCombo->getSelectedId();
326 const bool isContinuous = (typeId == 1);
327 const bool isDiscrete = (typeId == 2);
328
329 row.minEditor->setVisible(isContinuous);
330 row.maxEditor->setVisible(isContinuous);
331 row.stepEditor->setVisible(isContinuous);
332 row.stepNamesEditor->setVisible(isDiscrete);
333
334 resized();
335}

337void CustomParameterControlConfigDialog::removeParameterRow(int rowIndex)
338{
339 // Sync config from current UI state first, then remove entry
340 auto currentConfig = getConfig();
341 if (rowIndex < 0 || rowIndex >= int(currentConfig.parameters.size()))
342 return;
343 currentConfig.parameters.erase(currentConfig.parameters.begin() + rowIndex);
344
345 m_config = currentConfig;
346 rebuildParameterRows();
347 resized();
348}

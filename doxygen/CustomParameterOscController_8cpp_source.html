45bool CustomParameterOscController::connect()
46{
47 disconnect();
48
49 if (m_config.parameters.empty())
50 return true;
51
52 bool senderOk = m_sender.connect(m_config.oscRemoteHost, m_config.oscSendPort);
53 bool receiverOk = (m_config.oscReceivePort > 0)
54 && m_receiver.connect(m_config.oscReceivePort);
55
56 m_connected = senderOk;
57 return senderOk && receiverOk;
58}

67void CustomParameterOscController::sendParameterValue(int parameterIndex, float nativeValue)
68{
69 if (!m_connected)
70 return;
71 if (parameterIndex < 0 || parameterIndex >= int(m_config.parameters.size()))
72 return;
73
74 const auto& entry = m_config.parameters[parameterIndex];
75 if (entry.oscAddress.isEmpty())
76 return;
77
78 juce::OSCMessage msg(entry.oscAddress);
79 if (entry.controlType == JUCEAppBasics::ParameterControlType::Toggle)
80 msg.addInt32(nativeValue > 0.5f ? 1 : 0);
81 else
82 msg.addFloat32(nativeValue);
83 m_sender.send(msg);
84}

98void CustomParameterOscController::oscMessageReceived(const juce::OSCMessage& message)
99{
100 auto address = message.getAddressPattern().toString();
101 if (message.isEmpty())
102 return;
103
104 float value = 0.0f;
105 if (message[0].isFloat32())
106 value = message[0].getFloat32();
107 else if (message[0].isInt32())
108 value = float(message[0].getInt32());
109 else
110 return;
111
112 for (int i = 0; i < int(m_config.parameters.size()); ++i)
113 {
114 if (m_config.parameters[i].oscAddress == address)
115 {
116 m_pendingValues[i] = value;
117 triggerAsyncUpdate();
118 break;
119 }
120 }
121}

123void CustomParameterOscController::handleAsyncUpdate()
124{
125 if (!onParameterValueReceived)
126 return;
127
128 for (auto& [index, value] : m_pendingValues)
129 onParameterValueReceived(index, value);
130
131 m_pendingValues.clear();
132}

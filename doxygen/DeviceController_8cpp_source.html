52void DeviceController::handleMessage(const juce::Message& message)
53{
54 if (auto const scm = dynamic_cast<const StateChangeMessage*>(&message))
55 {
56 if (onStateChanged)
57 onStateChanged(scm->getState());
58 }
59 else if (auto const rorm = dynamic_cast<const RemoteObjectReceivedMessage*>(&message))
60 {
61 if (onRemoteObjectReceived)
62 onRemoteObjectReceived(rorm->getRemoteObject());
63 }
64}

67bool DeviceController::connect()
68{
69 if (getState() != State::Disconnected)
70 {
71 DBG(juce::String(__FUNCTION__) << " - nothing to do as we're not disconnected");
72 return false;
73 }
74 DBG(__FUNCTION__);
75 NanoOcp1::SoundscapeController::connect(m_ocp1IPAddress.toString().toStdString(), m_ocp1Port, m_ocp1Timeout);
76 return true;
77}

85void DeviceController::setConnectionParameters(juce::IPAddress ip, int port, int timeoutMs)
86{
87 DBG(juce::String(__FUNCTION__) << " " << ip.toString() << ":" << port);
88 m_ocp1IPAddress = ip;
89 m_ocp1Port = port;
90 jassert(0 < timeoutMs);

118bool DeviceController::SetActiveRemoteObjects(const std::vector<RemoteObject>& remObjs)
119{
120 std::vector<NanoOcp1::SoundscapeController::RemoteObject> ctrlObjs;
121 ctrlObjs.reserve(remObjs.size());

127const std::vector<DeviceController::RemoteObject>& DeviceController::GetActiveRemoteObjects()
128{
129 // Sync our cache from the base class on every call.
130 auto const& base = NanoOcp1::SoundscapeController::getActiveRemoteObjects();
131 m_activeRemoteObjects.clear();
132 m_activeRemoteObjects.reserve(base.size());
133 for (auto const& o : base)
134 m_activeRemoteObjects.push_back(fromCtrlObj(o));
135 return m_activeRemoteObjects;

74void UmsciUpmixSettingsComponent::resized()
75{
76 auto bounds = getLocalBounds().reduced(kMargin);
77
78 auto layoutRow = [&](juce::Label& label, juce::Component& control)
79 {
80 auto row = bounds.removeFromTop(kRowH);
81 label.setBounds(row.removeFromLeft(kLabelW));
82 control.setBounds(row);
83 bounds.removeFromTop(kRowGap);
84 };
85
86 layoutRow(m_controlFormatLabel, m_controlFormatCombo);
87 layoutRow(m_liveModeLabel, m_liveModeCombo);
88 layoutRow(m_shapeLabel, m_shapeCombo);
89 layoutRow(m_visualizationLabel, m_visualizationCombo);
90 layoutRow(m_startIdLabel, m_startIdEditor);
91 layoutRow(m_showSourcesLabel, m_showSourcesCombo);
92 layoutRow(m_lfeLabel, m_lfeCombo);
93 layoutRow(m_levelMeterLabel, m_levelMeterCombo);
94 layoutRow(m_channelLabelsLabel, m_channelLabelsCombo);
95}

34void CustomParameterControlComponent::setCustomConfig(const CustomParameterConfig& config)
35{
36 if (m_config == config)
37 return;
38
39 m_config = config;
40
41 std::vector<JUCEAppBasics::ParameterControlInfo> infos;
42 infos.reserve(config.parameters.size());
43 for (int i = 0; i < int(config.parameters.size()); ++i)
44 infos.push_back(config.parameters[i].toParameterControlInfo(i));
45
46 setParameters(infos);
47}
